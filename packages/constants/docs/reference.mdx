## Country Codes

```ts
import { COUNTRY_CODES, type CountryCode, isValidCountryCode } from "@aspen-os/constants";

const code: CountryCode = "US";

function setCountry(input: string): CountryCode {
  if (!isValidCountryCode(input)) {
    throw new Error(`Unknown country code: ${input}`);
  }
  return input; // narrowed to CountryCode
}

COUNTRY_CODES; // readonly ["AD", "AE", ..., "ZW"] — 249 ISO 3166-1 alpha-2 codes
```

<Callout type="info">
  `src/country-codes.ts` and `src/languages.ts` are empty placeholder files reserved for future
  expansion of country and language data beyond the alpha-2 codes in `COUNTRY_CODES`.
</Callout>
