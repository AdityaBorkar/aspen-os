The Management Plane declares its ACL via `defineAcl()` merged by the platform during `$prepareInfra()`.

## Resources & Operations

```ts title="src/auth.ts"
import { defineAcl } from "@aspen-os/platform/server";

export const acl = defineAcl({
  organization: ["create", "delete", "read", "update"],
  platformUser: ["create", "delete", "read", "update"],
  serviceProvider: ["create", "delete", "read", "update"],
  tenant: ["create", "delete", "read", "update"],
});
```

| Resource          | Operations                           |
| ----------------- | ------------------------------------ |
| `tenant`          | `create`, `read`, `update`, `delete` |
| `serviceProvider` | `create`, `read`, `update`, `delete` |
| `platformUser`    | `create`, `read`, `update`, `delete` |
| `organization`    | `create`, `read`, `update`, `delete` |

## Wiring

```ts title="src/module.ts"
$prepareInfra(): ModuleInfra {
  return {
    auth: { acl },
    db: { control_plane_schemas, tenant_schemas },
    events,
  };
}
```
